@router.get("/api/events/{company_id}")
async def sse_events(company_id: str):
    async def event_stream():
        q = subscribe(company_id)
        loop = asyncio.get_event_loop()
        try:
            while True:
                try:
                    ev = await loop.run_in_executor(
                        None, lambda: q.get(timeout=25)
                    )
                    yield f"data: {json.dumps(ev, ensure_ascii=False)}\n\n"
                except Exception:
                    yield 'data: {"type":"keepalive"}\n\n'
        finally:
            unsubscribe(company_id, q)

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "X-Accel-Buffering": "no",
        },
    )


@router.get("/api/metrics/{company_id}")
def get_metrics(company_id: str, days: int = 7):
    empty_payload = {
        "interactions_per_day": [],
        "tts_share": {},
        "age_range": {},
        "avg_latency": {},
    }

    if not os.path.exists(METRICS_CSV):
        return JSONResponse(empty_payload)

    df = pd.read_csv(METRICS_CSV)

    if "company_id" in df.columns:
        df = df[df["company_id"] == company_id]

    if df.empty:
        return JSONResponse(empty_payload)

    if "timestamp" in df.columns:
        df["date"] = pd.to_datetime(df["timestamp"], errors="coerce").dt.date.astype(str)

    interactions_per_day = (
        df.groupby("date").size().reset_index(name="count").tail(days).to_dict(orient="records")
        if "date" in df.columns else []
    )

    tts_share = df["voice_source"].value_counts().to_dict() if "voice_source" in df.columns else {}

    age_counts: Dict[str, int] = {}
    if "profile" in df.columns:
        for profile in df["profile"].dropna().tolist():
            age_range = None

            if isinstance(profile, dict):
                age_range = profile.get("age_range")
            else:
                text = str(profile)
                if "age_range" in text:
                    import re
                    match = re.search(r"age_range['\"]?\s*[:=]\s*['\"]([^'\"]+)['\"]", text)
                    if match:
                        age_range = match.group(1)

            if age_range:
                age_counts[age_range] = age_counts.get(age_range, 0) + 1

    avg_latency = {}
    if "gen_latency_s" in df.columns:
        avg_latency["gen"] = float(df["gen_latency_s"].dropna().mean())
    if "tts_latency_s" in df.columns:
        avg_latency["tts"] = float(df["tts_latency_s"].dropna().mean())

    return JSONResponse({
        "interactions_per_day": interactions_per_day,
        "tts_share": tts_share,
        "age_range": age_counts,
        "avg_latency": avg_latency,
    })

<script>
let FAQ = [];
let CANDIDATES = [];

function escapeHtml(value) {
  return String(value ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

async function loadData() {
  const faqRes = await fetch("/admin/faq/items");
  const candRes = await fetch("/admin/faq/candidates");

  FAQ = (await faqRes.json()).items || [];
  CANDIDATES = (await candRes.json()).items || [];

  renderAll();
}

function currentFilter() {
  return document.getElementById("search").value.trim().toLowerCase();
}

function matches(item) {
  const q = currentFilter();
  if (!q) return true;

  return (
    String(item.question || "").toLowerCase().includes(q) ||
    String(item.answer || "").toLowerCase().includes(q) ||
    String(item.intent || "").toLowerCase().includes(q)
  );
}

function renderAll() {
  renderCandidates();
  renderFaq();
}

function renderCandidates() {
  const list = CANDIDATES.filter(matches);
  const el = document.getElementById("candidates");

  if (!list.length) {
    el.innerHTML = `<div class="empty-state">Nenhum candidato pendente.</div>`;
    return;
  }

  el.innerHTML = list.map((item, index) => `
    <article class="mini-card">
      <div class="mini-card-title">${escapeHtml(item.question)}</div>
      <p>${escapeHtml(item.answer)}</p>
      <div class="badge">${escapeHtml(item.intent || "geral")}</div>
      <div class="actions" style="margin-top:10px;">
        <button class="btn primary" onclick="approveCandidate(${index})">Aprovar</button>
        <button class="btn danger" onclick="rejectCandidate(${index})">Rejeitar</button>
      </div>
    </article>
  `).join("");
}

function renderFaq() {
  const list = FAQ.filter(matches);
  const el = document.getElementById("faqItems");

  if (!list.length) {
    el.innerHTML = `<div class="empty-state">Nenhum FAQ aprovado encontrado.</div>`;
    return;
  }

  el.innerHTML = list.map((item, index) => `
    <article class="mini-card">
      <div class="mini-card-title">${escapeHtml(item.question)}</div>
      <p>${escapeHtml(item.answer)}</p>
      <div class="badge">${escapeHtml(item.intent || "geral")}</div>
      <div class="actions" style="margin-top:10px;">
        <button class="btn danger" onclick="deleteFaq(${index})">Excluir</button>
      </div>
    </article>
  `).join("");
}

async function createFaq() {
  const payload = {
    question: document.getElementById("faqQuestion").value,
    answer: document.getElementById("faqAnswer").value,
    intent: document.getElementById("faqIntent").value
  };

  const res = await fetch("/admin/faq/create", {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify(payload)
  });

  if (!res.ok) {
    alert("Erro ao salvar FAQ.");
    return;
  }

  document.getElementById("faqQuestion").value = "";
  document.getElementById("faqAnswer").value = "";

  await loadData();
}

async function approveCandidate(index) {
  const item = CANDIDATES.filter(matches)[index];

  await fetch("/admin/faq/approve", {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify(item)
  });

  await loadData();
}

async function rejectCandidate(index) {
  const item = CANDIDATES.filter(matches)[index];

  await fetch("/admin/faq/reject", {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify(item)
  });

  await loadData();
}

async function deleteFaq(index) {
  const item = FAQ.filter(matches)[index];

  if (!confirm(`Excluir FAQ: ${item.question}?`)) return;

  await fetch("/admin/faq/items", {
    method: "DELETE",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify({question: item.question})
  });

  await loadData();
}

loadData();
</script>

const companyId = "{{ company_id }}";

  document.getElementById("title").textContent = title;

  if (answer !== undefined) {
    document.getElementById("answer").textContent = answer;

async function activateTotem() {
  if (activated) return;
  activated = true;

  setStatus("PresenÃ§a detectada", "Ativando atendimento...", "Preparando saudaÃ§Ã£o...");

  try {
    const res = await fetch("/totem/activate", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({
        company_id: companyId,
        session_id: sessionId,
        prefer_audio: true
      })
    });

    if (!res.ok) {
      throw new Error("Falha HTTP em /totem/activate: " + res.status);
    }

    const data = await res.json();

    document.getElementById("answer").textContent = data.greeting || "OlÃ¡! Como posso ajudar hoje?";
    enableUI();

    if (data.audio_base64) {
      currentAudio = new Audio("data:audio/mp3;base64," + data.audio_base64);
      currentAudio.play().catch(() => {});
    }
  } catch (error) {
    console.error("Erro ao ativar totem:", error);
    document.getElementById("answer").textContent = "OlÃ¡! Como posso ajudar hoje?";
    enableUI();
  }
}

function connectSSE() {
  eventSource = new EventSource(`/api/events/${companyId}`);

  eventSource.onopen = function() {
    console.log("SSE conectado");
  };

  eventSource.onmessage = function(event) {
    try {
      const data = JSON.parse(event.data);
      console.log("SSE recebido:", data);

      if (data.type === "presence_triggered") {
        activateTotem();
      }
    } catch (error) {
      console.error("Erro ao processar SSE:", error);
    }
  };

  eventSource.onerror = function(error) {
    console.warn("Erro SSE:", error);
  };
}

import os

AUDIO_PATH = Path("/tmp/input.wav")

ARECORD_FORMAT = os.getenv("ARECORD_FORMAT", "cd")


def record_audio(seconds: int = 4) -> str | None:
    print("[AUDIO] gravando...")

    if AUDIO_PATH.exists():
        AUDIO_PATH.unlink()

    cmd = [
        "arecord",
        "-D",
        ARECORD_DEVICE,
        "-f",
        ARECORD_FORMAT,
        "-d",
        str(seconds),
        str(AUDIO_PATH),
    ]

    try:
        result = subprocess.run(
            cmd,
            check=False,
            capture_output=True,
            text=True,
        )
    except FileNotFoundError:
        print("[ERRO] arecord nÃ£o estÃ¡ instalado")
        return None
    except Exception as exc:
        print(f"[ERRO] falha ao iniciar gravaÃ§Ã£o: {exc}")
        return None

    if result.returncode != 0:
        print("[ERRO] falha ao gravar Ã¡udio")
        if result.stderr:
            print(result.stderr.strip())
        return None

    if not AUDIO_PATH.exists() or AUDIO_PATH.stat().st_size == 0:
        print("[ERRO] arquivo de Ã¡udio nÃ£o foi criado corretamente")
        return None

    print("[AUDIO] gravaÃ§Ã£o concluÃ­da")
    return str(AUDIO_PATH)

import base64
import os
import subprocess
import time

from config import CAMERA_INDEX, CAMERA_WIDTH, CAMERA_HEIGHT, JPEG_QUALITY

OUTPUT_PATH = "/tmp/last_capture.jpg"
LOCK_PATH = "/tmp/camera.lock"


def _acquire_lock():
    if os.path.exists(LOCK_PATH):
        return False
    with open(LOCK_PATH, "w") as f:
        f.write("1")
    return True


def _release_lock():
    if os.path.exists(LOCK_PATH):
        os.remove(LOCK_PATH)


def capture_image_base64() -> str | None:
    if not _acquire_lock():
        print("[WARN] cÃ¢mera ocupada (lock ativo)")
        return None

    try:
        device = f"/dev/video{CAMERA_INDEX}"

        if os.path.exists(OUTPUT_PATH):
            os.remove(OUTPUT_PATH)

        time.sleep(0.5)

        cmd = [
            "fswebcam",
            "-d", device,
            "-r", f"{CAMERA_WIDTH}x{CAMERA_HEIGHT}",
            "--jpeg", str(JPEG_QUALITY),
            "--no-banner",
            OUTPUT_PATH,
        ]

        print(f"[DEBUG] executando: {' '.join(cmd)}")

        result = subprocess.run(
            cmd,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            timeout=5
        )

        if result.returncode != 0:
            print("[ERRO] fswebcam falhou")
            return None

        if not os.path.exists(OUTPUT_PATH):
            print("[ERRO] imagem nÃ£o foi criada")
            return None

        with open(OUTPUT_PATH, "rb") as f:
            return base64.b64encode(f.read()).decode("utf-8")

    except Exception as e:
        print(f"[ERRO cÃ¢mera]: {e}")
        return None

    finally:
        time.sleep(1.5)
        _release_lock()

from __future__ import annotations

import time

from config import COOLDOWN_SECONDS, PIR_PIN, PRESENCE_HOLD_SECONDS
from pir_sensor import read_motion
from sender import send_trigger

SAMPLE_INTERVAL_SECONDS = 0.1
MIN_ACTIVE_RATIO = 0.35
WARMUP_SECONDS = 5


def confirm_presence_window() -> bool:
    total_samples = max(1, int(PRESENCE_HOLD_SECONDS / SAMPLE_INTERVAL_SECONDS))
    active_samples = 0

    start = time.time()
    while time.time() - start < PRESENCE_HOLD_SECONDS:
        if read_motion():
            active_samples += 1
        time.sleep(SAMPLE_INTERVAL_SECONDS)

    ratio = active_samples / total_samples

    print(
        f"Janela presenÃ§a | ativos={active_samples} "
        f"total={total_samples} ratio={ratio:.2f}"
    )

    return ratio >= MIN_ACTIVE_RATIO


def main() -> None:
    print("Presence sender iniciado")
    print(f"PIR_PIN={PIR_PIN}")
    print(f"HOLD={PRESENCE_HOLD_SECONDS}s")
    print(f"COOLDOWN={COOLDOWN_SECONDS}s")
    print(f"Aguardando estabilizaÃ§Ã£o do PIR por {WARMUP_SECONDS}s...")

    time.sleep(WARMUP_SECONDS)

    print("Sensor estabilizado. Monitorando presenÃ§a...")

    last_trigger_time = 0.0

    while True:
        try:
            now = time.time()

            if not read_motion():
                time.sleep(0.2)
                continue

            if now - last_trigger_time < COOLDOWN_SECONDS:
                time.sleep(0.2)
                continue

            print("Movimento detectado - validando janela do PIR...")

            if not confirm_presence_window():
                print("PresenÃ§a descartada - janela insuficiente.")
                time.sleep(0.2)
                continue

            print("PresenÃ§a contÃ­nua confirmada - enviando trigger...")

            if send_trigger():
                print("Trigger aceito pela API.")
                last_trigger_time = time.time()
            else:
                print("Trigger rejeitado ou falhou.")

            time.sleep(0.2)

        except KeyboardInterrupt:
            print("Encerrado manualmente.")
            break

        except Exception as exc:
            print(f"[ERRO LOOP]: {exc}")
            time.sleep(1)


if __name__ == "__main__":
    main()

)

def send_audio(file_path: str) -> str | None:
    print("[AUDIO] enviando para transcriÃ§Ã£o LOCAL...")

    try:
        with open(file_path, "rb") as audio_file:
            files = {"file": ("input.wav", audio_file, "audio/wav")}
            response = requests.post(
                API_URL,
                files=files,
                timeout=REQUEST_TIMEOUT,
            )

        if response.status_code != 200:
            print(f"[ERRO] falha na API: {response.status_code}")
            print(response.text)
            return None

        data = response.json()
        text = (data.get("text") or "").strip()

        print(f"[TRANSCRITO] {text}")

        return text

    except Exception as exc:
        print(f"[ERRO] transcriÃ§Ã£o: {exc}")
        return None

import requests

from camera import capture_image_base64
from config import (
    TOTEM_API_URL,
    COMPANY_ID,
    DEVICE_ID,
    REQUEST_TIMEOUT_SECONDS,
)


def send_trigger() -> bool:
    if not TOTEM_API_URL:
        print("Erro: TOTEM_API_URL nÃ£o configurada")
        return False

    image_base64 = capture_image_base64()
    if not image_base64:
        print("Erro: imagem nÃ£o capturada; trigger cancelado")
        return False

    payload = {
        "company_id": COMPANY_ID,
        "device_id": DEVICE_ID,
        "image_base64": image_base64,
    }

    try:
        response = requests.post(
            TOTEM_API_URL,
            json=payload,
            timeout=REQUEST_TIMEOUT_SECONDS,
        )

        print(f"Status: {response.status_code}")
        print(response.text)

        if response.status_code != 200:
            return False

        data = response.json()
        state = data.get("state", {})
        accepted = bool(state.get("present", False))

        if accepted:
            print("PresenÃ§a aceita pela AWS")
        else:
            print("PresenÃ§a rejeitada pela AWS")

        return accepted

    except Exception as exc:
        print(f"Erro ao enviar trigger: {exc}")
        return False
